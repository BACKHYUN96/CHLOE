cmake_minimum_required(VERSION 3.22)
project(chloe_llm LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-g0)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
add_subdirectory("../../../../external/llama.cpp" llama-build)
add_library(chloe_llm SHARED llm_jni.cpp)
target_link_libraries(chloe_llm PRIVATE llama-common llama log)
target_link_options(chloe_llm PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,-z,max-page-size=16384")
